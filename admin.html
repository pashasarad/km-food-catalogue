ALTER TABLE public.product ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Public can read products" ON public.product; DROP POLICY IF EXISTS "Authenticated can read products" ON public.product; DROP POLICY IF EXISTS "Authenticated can insert products" ON public.product; DROP POLICY IF EXISTS "Authenticated can update products" ON public.product; DROP POLICY IF EXISTS "admin_manage_product" ON public.product; DROP POLICY IF EXISTS "public_read_product" ON public.product; /* Everyone can READ product ON/OFF status */ CREATE POLICY "public_read_product" ON public.product FOR SELECT TO anon, authenticated USING (true); /* Logged-in admin can INSERT / UPDATE / DELETE */ CREATE POLICY "admin_manage_product" ON public.product FOR ALL TO authenticated USING (true) WITH CHECK (true);
